Reusable Razor partial: every field gets a label, hint, and an error message wired up with aria-describedby.

@model FieldVm
@{ var err = ViewData.ModelState[Model.Name]?.Errors.FirstOrDefault()?.ErrorMessage; }
<div class="field">
  <label for="@Model.Id">@Model.Label</label>
  @if (Model.Hint != null) { <span class="hint" id="@(Model.Id)-hint">@Model.Hint</span> }
  <input id="@Model.Id" name="@Model.Name" type="@Model.Type"
         value="@Model.Value"
         aria-required="@Model.Required.ToString().ToLower()"
         aria-invalid="@(err != null ? "true" : "false")"
         aria-describedby="@(Model.Hint != null ? Model.Id + "-hint " : "")@(err != null ? Model.Id + "-err" : "")" />
  @if (err != null) { <div class="msg" id="@(Model.Id)-err">@err</div> }
</div>

ASP.NET Core Web API: thin controller, validation in the service, field-level errors with no internals exposed.

[ApiController, Route("api/licenses")]
public class LicensesController(ILicenseService svc) : ControllerBase
{
    [HttpPost("{id:int}/renew")]
    [ProducesResponseType(typeof(RenewalReceipt), 200)]
    [ProducesResponseType(typeof(ValidationProblemDetails), 400)]
    public async Task<IActionResult> Renew(int id, RenewalRequest req, CancellationToken ct)
    {
        var result = await svc.RenewAsync(id, req, ct);
        if (!result.IsValid)
        {
            foreach (var e in result.Errors)
                ModelState.AddModelError(e.Field, e.Message);
            return ValidationProblem(ModelState);
        }
        return Ok(result.Receipt);
    }
}

The pattern used by the live form above: collect errors, show a linked summary, move focus to it.

function showErrors(errors) {            // errors: [{ id, text }]
  const box = document.getElementById('summary');
  box.innerHTML = '<h3>Fix ' + errors.length + ' problem(s) to continue</h3><ul>' +
    errors.map(e => '<li><a href="#' + e.id + '">' + e.text + '</a></li>').join('') + '</ul>';
  box.className = 'summary';
  box.hidden = false;
  box.focus();                           // screen readers announce the summary
}
// Each summary link focuses its field instead of only jumping the page:
box.addEventListener('click', ev => {
  const a = ev.target.closest('a'); if (!a) return;
  ev.preventDefault();
  document.getElementById(a.hash.slice(1)).focus();
});

Azure DevOps pipeline: restore, build, test, publish an artifact on every push to main.

trigger: [ main ]
pool: { vmImage: windows-latest }
steps:
- task: UseDotNet@2
  inputs: { version: '10.x' }
- script: dotnet restore
- script: dotnet build --configuration Release --no-restore
- script: dotnet test --configuration Release --no-build --logger trx
- task: PublishTestResults@2
  inputs: { testResultsFormat: VSTest, testResultsFiles: '**/*.trx' }
- script: dotnet publish src/Web -c Release -o $(Build.ArtifactStagingDirectory)
- task: PublishBuildArtifacts@1

How I build for accessibility

  • Semantic HTML first; ARIA only where HTML can’t say it.
  • Every control reachable and operable by keyboard, with a visible focus ring.
  • Logical focus order, and focus moved on purpose after errors or dialogs.
  • Colour contrast checked for text and controls in light and dark themes.
  • Labels, hints and errors programmatically linked to their fields.
  • Automated checks with axe and Lighthouse, then manual testing with JAWS and NVDA.
  • Motion respects the reduced-motion setting; layouts reflow on small screens.
  • Touch targets of at least 44 px.
